{

	//Active->SLEEP->ACTIVE
	printf("ACTIVE STATE TO LPSLEPPING STATE\n");
	//============================================================
	//   Initializtion
	//============================================================
	sec_wr(STATUS_REG_0, 0x0);


	sec_wr(I2BZ_AON_SEC_SYSREG_PCR_CM4_DN_CNT0_MAX,3);
	sec_wr(I2BZ_AON_SEC_SYSREG_PCR_CM4_DN_CNT1_MAX,15);
	sec_wr(I2BZ_AON_SEC_SYSREG_PCR_CM4_UP_CNT0_MAX,3);
	sec_wr(I2BZ_AON_SEC_SYSREG_PCR_CM4_UP_CNT1_MAX,15);
	sec_wr(I2BZ_AON_SEC_SYSREG_PCR_CM4_RST_REL_CNT_MAX,15);

	rd_data = sec_rd(I2BZ_AON_SEC_SYSREG_SCRATCH_REG0);
	if(rd_data != 0x12345678)
	{
		sec_wr(I2BZ_AON_SEC_SYSREG_SCRATCH_REG0,0x12345678);
		wait_on(10000);
		sec_wr(I2BZ_AON_SEC_SYSREG_MISC,0x67);
		//============================================================
		//Configuring the BAR register
		//============================================================
		sec_wr(I2BW_PACMAN0_BAR_0, PRELOAD_ADDR_LSB);
		sec_wr(I2BW_PACMAN0_BAR_OFFSET_BASE, PRELOAD_ADDR_MSB);
		//============================================================
		//Starting the Pacman 
		//============================================================
		sec_wr(I2BW_PACMAN0_PACMAN_CONTROL_REG, 0x1);
		//============================================================
		//Pacman master is now doing secure access 
		//============================================================
		sec_wr(I2BW_PERI_SEC_SYSREG_CM4_PACMAN_SEC_ACCESS, 0x1);

		//	intr = 1;
		//Enabling intr line i to vector0 
		sec_wr(I2BW_PACMAN0_INTCTRL_23, 0x1); 
		sec_wr(I2BZ_AON_SEC_SYSREG_PCR_CM4_REG,0x1);
		wait_on(90000);	
	}
	finish(0);

}
